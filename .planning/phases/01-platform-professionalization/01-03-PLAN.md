    2. Export the logError function:
    ```typescript
    export function logError(error: unknown, context: LogContext): void {
      const entry = {
        timestamp: new Date().toISOString(),
        operation: context.operation,
        userId: context.userId ?? "anonymous",
        error: error instanceof Error
          ? { message: error.message, stack: error.stack }
          : String(error),
        ...(context.extra ?? {}),
      };
      console.error("[arcline]", entry);
    }
    ```

    No other exports. No default export. No imports needed.

    The `...(context.extra ?? {})` spread puts extra fields at the top level of the entry object (not nested under "extra") — this makes log parsing simpler and matches the research design.

    Do NOT use JSON.stringify in the console.error call — pass the object directly so browser DevTools and test spies can inspect the object structure natively.
